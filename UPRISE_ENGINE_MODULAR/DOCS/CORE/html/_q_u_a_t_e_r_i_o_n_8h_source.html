346 __inline UPRISE_CORE_API Vector3 ToRotationVector() const {
347
348
349
350 float length = std::sqrt(w() * w() + x() * x() + y() * y() + z() * z());
351 float nw = w() / length;
352 float nx = x() / length;
353 float ny = y() / length;
354 float nz = z() / length;
355
356 // Compute the angle
357 float angle = 2.0f * std::acos(nw);
358
359 // Compute the axis
360 float s = std::sqrt(1.0f - nw * nw);
361 if (s < 0.0001f) {
362 // If s is close to zero, return the axis as (1, 0, 0)
363 return Vector3(1, 0, 0) * angle;
364 }
365 else {
366 return Vector3(nx / s, ny / s, nz / s) * angle;
367 }
368 }

371__inline UPRISE_CORE_API Vector3 operator*(Vector3 vec) {
372 float num = x() * 2;
373 float num2 = y() * 2;
374 float num3 = z() * 2;
375 float num4 = x() * num;
376 float num5 = y() * num2;
377 float num6 = z() * num3;
378 float num7 = x() * num2;
379 float num8 = x() * num3;
380 float num9 = y() * num3;
381 float num10 = w() * num;
382 float num11 = w() * num2;
383 float num12 = w() * num3;
384 Vector3 result;
385 result.x() = (1 - (num5 + num6)) * vec.x() + (num7 - num12) * vec.y() + (num8 + num11) * vec.z();
386 result.y() = (num7 + num12) * vec.x() + (1 - (num4 + num6)) * vec.y() + (num9 - num10) * vec.z();
387 result.z() = (num8 - num11) * vec.x() + (num9 + num10) * vec.y() + (1 - (num4 + num5)) * vec.z();
388 return result;
389 }
