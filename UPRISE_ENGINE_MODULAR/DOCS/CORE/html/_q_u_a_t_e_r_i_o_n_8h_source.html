213 __inline UPRISE_CORE_API Vector3 ToRotationVector() const {
214
215
216
217 float length = std::sqrt(w() * w() + x() * x() + y() * y() + z() * z());
218 float nw = w() / length;
219 float nx = x() / length;
220 float ny = y() / length;
221 float nz = z() / length;
222
223 // Compute the angle
224 float angle = 2.0f * std::acos(nw);
225
226 // Compute the axis
227 float s = std::sqrt(1.0f - nw * nw);
228 if (s < 0.0001f) {
229 // If s is close to zero, return the axis as (1, 0, 0)
230 return Vector3(1, 0, 0) * angle;
231 }
232 else {
233 return Vector3(nx / s, ny / s, nz / s) * angle;
234 }
