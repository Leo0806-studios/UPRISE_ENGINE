69 void Mesh::setupMesh()
70 {
71
72 glGenVertexArrays(1, &VAO);
73 glGenBuffers(1, &VBO);
74 glGenBuffers(1, &EBO);
75 glBindVertexArray(VAO);
76 glBindBuffer(GL_ARRAY_BUFFER, VBO);
77 glBufferData(GL_ARRAY_BUFFER, vertices.size() * sizeof(Vertex),
78 &vertices[0], GL_STATIC_DRAW);
79 glBindBuffer(GL_ELEMENT_ARRAY_BUFFER, EBO);
80 glBufferData(GL_ELEMENT_ARRAY_BUFFER, indices.size() *
81 sizeof(unsigned int), &indices[0], GL_STATIC_DRAW);
82 // vertex positions
83 glEnableVertexAttribArray(0);
84 glVertexAttribPointer(0, 3, GL_FLOAT, GL_FALSE, sizeof(Vertex),
85 (void*)0);
86 // vertex normals
87 glEnableVertexAttribArray(1);
88 glVertexAttribPointer(1, 3, GL_FLOAT, GL_FALSE, sizeof(Vertex),
89 (void*)offsetof(Vertex, Normal));
90 // vertex texture coords
91 glEnableVertexAttribArray(2);
92 glVertexAttribPointer(2, 2, GL_FLOAT, GL_FALSE, sizeof(Vertex),
93 (void*)offsetof(Vertex, TexCoords));
94 glBindVertexArray(0);

259 unsigned int Model::TextureFromFile(const char* path, const string& directory, bool gamma)
260 {
261
262 string filename = string(path);
263 filename = directory + '/' + filename;
264
265 unsigned int textureID;
266 glGenTextures(1, &textureID);
267
268 int width, height, nrComponents;
269 unsigned char* data = stbi_load(filename.c_str(), &width, &height, &nrComponents, 0);
270 if (data)
271 {
272 GLenum format;
273 if (nrComponents == 1)
274 format = GL_RED;
275 else if (nrComponents == 3)
276 format = GL_RGB;
277 else if (nrComponents == 4)
278 format = GL_RGBA;
279
280 glBindTexture(GL_TEXTURE_2D, textureID);
281 glTexImage2D(GL_TEXTURE_2D, 0, format, width, height, 0, format, GL_UNSIGNED_BYTE, data);
282 glGenerateMipmap(GL_TEXTURE_2D);
283
284 glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_WRAP_S, GL_REPEAT);
285 glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_WRAP_T, GL_REPEAT);
286 glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_MIN_FILTER, GL_LINEAR_MIPMAP_LINEAR);
287 glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_MAG_FILTER, GL_LINEAR);
288
289 stbi_image_free(data);
290 }
291 else
292 {
293 std::cout << "Texture failed to load at path: " << path << std::endl;
294 stbi_image_free(data);
295 }
296
297 return textureID;
298 }
