518 // Updates velocity and orientation by tracking changes in position over time.
519 void XM_CALLCONV Update(FXMVECTOR newPos, XMVECTOR upDir, float dt) noexcept
520 {
521 if (dt > 0.f)
522 {
523 const XMVECTOR lastPos = XMLoadFloat3(reinterpret_cast<const XMFLOAT3*>(&Position));
524
525 XMVECTOR vDelta = XMVectorSubtract(newPos, lastPos);
526 const XMVECTOR vt = XMVectorReplicate(dt);
527 XMVECTOR v = XMVectorDivide(vDelta, vt);
528 XMStoreFloat3(reinterpret_cast<XMFLOAT3*>(&Velocity), v);
529
530 vDelta = XMVector3Normalize(vDelta);
531 XMStoreFloat3(reinterpret_cast<XMFLOAT3*>(&OrientFront), vDelta);
532
533 v = XMVector3Cross(upDir, vDelta);
534 v = XMVector3Normalize(v);
535
536 v = XMVector3Cross(vDelta, v);
537 v = XMVector3Normalize(v);
538 XMStoreFloat3(reinterpret_cast<XMFLOAT3*>(&OrientTop), v);
539
540 XMStoreFloat3(reinterpret_cast<XMFLOAT3*>(&Position), newPos);
541 }
