248 // Helper for resource barrier.
249 inline void TransitionResource(
250 _In_ ID3D12GraphicsCommandList* commandList,
251 _In_ ID3D12Resource* resource,
252 D3D12_RESOURCE_STATES stateBefore,
253 D3D12_RESOURCE_STATES stateAfter) noexcept
254 {
255 assert(commandList != nullptr);
256 assert(resource != nullptr);
257
258 if (stateBefore == stateAfter)
259 return;
260
261 D3D12_RESOURCE_BARRIER desc = {};
262 desc.Type = D3D12_RESOURCE_BARRIER_TYPE_TRANSITION;
263 desc.Transition.pResource = resource;
264 desc.Transition.Subresource = D3D12_RESOURCE_BARRIER_ALL_SUBRESOURCES;
265 desc.Transition.StateBefore = stateBefore;
266 desc.Transition.StateAfter = stateAfter;
267
268 commandList->ResourceBarrier(1, &desc);
269 }

271 // Helper which applies one or more resources barriers and then reverses them on destruction.
272 class ScopedBarrier
273 {
274 public:
275 ScopedBarrier(
276 _In_ ID3D12GraphicsCommandList* commandList,
277 std::initializer_list<D3D12_RESOURCE_BARRIER> barriers) noexcept(false)
278 : mCommandList(commandList),
279 mBarriers(barriers)
280 {
281 assert(mBarriers.size() <= UINT32_MAX);
282
283 // Set barriers
284 mCommandList->ResourceBarrier(static_cast<UINT>(mBarriers.size()), mBarriers.data());
285 }
286
287 ScopedBarrier(
288 _In_ ID3D12GraphicsCommandList* commandList,
289 _In_reads_(count) const D3D12_RESOURCE_BARRIER *barriers,
290 size_t count) noexcept(false)
291 : mCommandList(commandList),
292 mBarriers(barriers, barriers + count)
293 {
294 assert(count <= UINT32_MAX);
295
296 // Set barriers
297 mCommandList->ResourceBarrier(static_cast<UINT>(mBarriers.size()), mBarriers.data());
298 }
299
300 template<size_t TBarrierLength>
301 ScopedBarrier(
302 _In_ ID3D12GraphicsCommandList* commandList,
303 const D3D12_RESOURCE_BARRIER(&barriers)[TBarrierLength]) noexcept(false)
304 : mCommandList(commandList),
305 mBarriers(barriers, barriers + TBarrierLength)
306 {
307 assert(TBarrierLength <= UINT32_MAX);
308
309 // Set barriers
310 mCommandList->ResourceBarrier(static_cast<UINT>(mBarriers.size()), mBarriers.data());
311 }
312
313 ScopedBarrier(ScopedBarrier&&) = default;
314 ScopedBarrier& operator= (ScopedBarrier&&) = default;
315
316 ScopedBarrier(ScopedBarrier const&) = delete;
317 ScopedBarrier& operator= (ScopedBarrier const&) = delete;
318
319 ~ScopedBarrier()
320 {
321 // reverse barrier inputs and outputs
322 for (auto& b : mBarriers)
323 {
324 std::swap(b.Transition.StateAfter, b.Transition.StateBefore);
325 }
326
327 // Set barriers
328 mCommandList->ResourceBarrier(static_cast<UINT>(mBarriers.size()), mBarriers.data());
329 }
330
331 private:
332 ID3D12GraphicsCommandList* mCommandList;
333 std::vector<D3D12_RESOURCE_BARRIER> mBarriers;
334 };

342 // Helpers for aligning values by a power of 2
343 template<typename T>
344 inline T AlignDown(T size, size_t alignment) noexcept
345 {
346 if (alignment > 0)
347 {
348 assert(((alignment - 1) & alignment) == 0);
349 auto mask = static_cast<T>(alignment - 1);
350 return size & ~mask;
351 }
352 return size;
353 }
354
355 template<typename T>
356 inline T AlignUp(T size, size_t alignment) noexcept
357 {
358 if (alignment > 0)
359 {
360 assert(((alignment - 1) & alignment) == 0);
361 auto mask = static_cast<T>(alignment - 1);
362 return (size + mask) & ~mask;
363 }
364 return size;
365 }
