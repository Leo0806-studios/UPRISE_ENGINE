284 // ---------------------------------------------------------------------
286 template <typename T>
287 T Get() {
288 if (mCurrent + sizeof(T) > mLimit) {
289 throw DeadlyImportError("End of file or stream limit was reached");
290 }
291
292 T f;
293 ::memcpy(&f, mCurrent, sizeof(T));
294 Intern::Getter<SwapEndianess, T, RuntimeSwitch>()(&f, mLe);
295 mCurrent += sizeof(T);
296
297 return f;
298 }
299
300private:
301 // ---------------------------------------------------------------------
302 void InternBegin() {
303 if (nullptr == mStream) {
304 throw DeadlyImportError("StreamReader: Unable to open file");
305 }
306
307 const size_t filesize = mStream->FileSize() - mStream->Tell();
308 if (0 == filesize) {
309 throw DeadlyImportError("StreamReader: File is empty or EOF is already reached");
310 }
311
312 mCurrent = mBuffer = new int8_t[filesize];
313 const size_t read = mStream->Read(mCurrent, 1, filesize);
314 // (read < s) can only happen if the stream was opened in text mode, in which case FileSize() is not reliable
315 ai_assert(read <= filesize);
316 mEnd = mLimit = &mBuffer[read - 1] + 1;
317 }
318
319private:
320 std::shared_ptr<IOStream> mStream;
321 int8_t *mBuffer;
322 int8_t *mCurrent;
323 int8_t *mEnd;
324 int8_t *mLimit;
325 bool mLe;
326};
