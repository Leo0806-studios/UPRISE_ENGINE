306AI_FORCE_INLINE bool IOSystem::CreateDirectory( const std::string &path ) {
307 if ( path.empty() ) {
308 return false;
309 }
310
311#ifdef _WIN32
312 return 0 != ::_mkdir( path.c_str() );
313#else
314 return 0 != ::mkdir( path.c_str(), 0777 );
315#endif // _WIN32
316}
