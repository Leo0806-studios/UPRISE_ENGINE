344// X11 does not include current pressed/released modifier key in 'mods' flags submitted by GLFW
345// See https://github.com/ocornut/imgui/issues/6034 and https://github.com/glfw/glfw/issues/1630
346static void ImGui_ImplGlfw_UpdateKeyModifiers(GLFWwindow* window)
347{
348 ImGuiIO& io = ImGui::GetIO();
349 io.AddKeyEvent(ImGuiMod_Ctrl, (glfwGetKey(window, GLFW_KEY_LEFT_CONTROL) == GLFW_PRESS) || (glfwGetKey(window, GLFW_KEY_RIGHT_CONTROL) == GLFW_PRESS));
350 io.AddKeyEvent(ImGuiMod_Shift, (glfwGetKey(window, GLFW_KEY_LEFT_SHIFT) == GLFW_PRESS) || (glfwGetKey(window, GLFW_KEY_RIGHT_SHIFT) == GLFW_PRESS));
351 io.AddKeyEvent(ImGuiMod_Alt, (glfwGetKey(window, GLFW_KEY_LEFT_ALT) == GLFW_PRESS) || (glfwGetKey(window, GLFW_KEY_RIGHT_ALT) == GLFW_PRESS));
352 io.AddKeyEvent(ImGuiMod_Super, (glfwGetKey(window, GLFW_KEY_LEFT_SUPER) == GLFW_PRESS) || (glfwGetKey(window, GLFW_KEY_RIGHT_SUPER) == GLFW_PRESS));
353}

389static int ImGui_ImplGlfw_TranslateUntranslatedKey(int key, int scancode)
390{
391#if GLFW_HAS_GETKEYNAME && !defined(EMSCRIPTEN_USE_EMBEDDED_GLFW3)
392 // GLFW 3.1+ attempts to "untranslate" keys, which goes the opposite of what every other framework does, making using lettered shortcuts difficult.
393 // (It had reasons to do so: namely GLFW is/was more likely to be used for WASD-type game controls rather than lettered shortcuts, but IHMO the 3.1 change could have been done differently)
394 // See https://github.com/glfw/glfw/issues/1502 for details.
395 // Adding a workaround to undo this (so our keys are translated->untranslated->translated, likely a lossy process).
396 // This won't cover edge cases but this is at least going to cover common cases.
397 if (key >= GLFW_KEY_KP_0 && key <= GLFW_KEY_KP_EQUAL)
398 return key;
399 GLFWerrorfun prev_error_callback = glfwSetErrorCallback(nullptr);
400 const char* key_name = glfwGetKeyName(key, scancode);
401 glfwSetErrorCallback(prev_error_callback);
402#if GLFW_HAS_GETERROR && !defined(EMSCRIPTEN_USE_EMBEDDED_GLFW3) // Eat errors (see #5908)
403 (void)glfwGetError(nullptr);
404#endif
405 if (key_name && key_name[0] != 0 && key_name[1] == 0)
406 {
407 const char char_names[] = "`-=[]\\,;\'./";
408 const int char_keys[] = { GLFW_KEY_GRAVE_ACCENT, GLFW_KEY_MINUS, GLFW_KEY_EQUAL, GLFW_KEY_LEFT_BRACKET, GLFW_KEY_RIGHT_BRACKET, GLFW_KEY_BACKSLASH, GLFW_KEY_COMMA, GLFW_KEY_SEMICOLON, GLFW_KEY_APOSTROPHE, GLFW_KEY_PERIOD, GLFW_KEY_SLASH, 0 };
409 IM_ASSERT(IM_ARRAYSIZE(char_names) == IM_ARRAYSIZE(char_keys));
410 if (key_name[0] >= '0' && key_name[0] <= '9') { key = GLFW_KEY_0 + (key_name[0] - '0'); }
411 else if (key_name[0] >= 'A' && key_name[0] <= 'Z') { key = GLFW_KEY_A + (key_name[0] - 'A'); }
412 else if (key_name[0] >= 'a' && key_name[0] <= 'z') { key = GLFW_KEY_A + (key_name[0] - 'a'); }
413 else if (const char* p = strchr(char_names, key_name[0])) { key = char_keys[p - char_names]; }
414 }
415 // if (action == GLFW_PRESS) printf("key %d scancode %d name '%s'\n", key, scancode, key_name);
416#else
417 IM_UNUSED(scancode);
418#endif
419 return key;
420}

511static EM_BOOL ImGui_ImplEmscripten_WheelCallback(int, const EmscriptenWheelEvent* ev, void*)
512{
513 // Mimic Emscripten_HandleWheel() in SDL.
514 // Corresponding equivalent in GLFW JS emulation layer has incorrect quantizing preventing small values. See #6096
515 float multiplier = 0.0f;
516 if (ev->deltaMode == DOM_DELTA_PIXEL) { multiplier = 1.0f / 100.0f; } // 100 pixels make up a step.
517 else if (ev->deltaMode == DOM_DELTA_LINE) { multiplier = 1.0f / 3.0f; } // 3 lines make up a step.
518 else if (ev->deltaMode == DOM_DELTA_PAGE) { multiplier = 80.0f; } // A page makes up 80 steps.
519 float wheel_x = ev->deltaX * -multiplier;
520 float wheel_y = ev->deltaY * -multiplier;
521 ImGuiIO& io = ImGui::GetIO();
522 io.AddMouseWheelEvent(wheel_x, wheel_y);
523 //IMGUI_DEBUG_LOG("[Emsc] mode %d dx: %.2f, dy: %.2f, dz: %.2f --> feed %.2f %.2f\n", (int)ev->deltaMode, ev->deltaX, ev->deltaY, ev->deltaZ, wheel_x, wheel_y);
524 return EM_TRUE;
525}

588static bool ImGui_ImplGlfw_Init(GLFWwindow* window, bool install_callbacks, GlfwClientApi client_api)
589{
590 ImGuiIO& io = ImGui::GetIO();
591 IMGUI_CHECKVERSION();
592 IM_ASSERT(io.BackendPlatformUserData == nullptr && "Already initialized a platform backend!");
593 //printf("GLFW_VERSION: %d.%d.%d (%d)", GLFW_VERSION_MAJOR, GLFW_VERSION_MINOR, GLFW_VERSION_REVISION, GLFW_VERSION_COMBINED);
594
595 // Setup backend capabilities flags
596 ImGui_ImplGlfw_Data* bd = IM_NEW(ImGui_ImplGlfw_Data)();
597 io.BackendPlatformUserData = (void*)bd;
598 io.BackendPlatformName = "imgui_impl_glfw";
599 io.BackendFlags |= ImGuiBackendFlags_HasMouseCursors; // We can honor GetMouseCursor() values (optional)
600 io.BackendFlags |= ImGuiBackendFlags_HasSetMousePos; // We can honor io.WantSetMousePos requests (optional, rarely used)
601#ifndef __EMSCRIPTEN__
602 io.BackendFlags |= ImGuiBackendFlags_PlatformHasViewports; // We can create multi-viewports on the Platform side (optional)
603#endif
604#if GLFW_HAS_MOUSE_PASSTHROUGH || GLFW_HAS_WINDOW_HOVERED
605 io.BackendFlags |= ImGuiBackendFlags_HasMouseHoveredViewport; // We can call io.AddMouseViewportEvent() with correct data (optional)
606#endif
607
608 bd->Window = window;
609 bd->Time = 0.0;
610 bd->WantUpdateMonitors = true;
611
612 io.SetClipboardTextFn = ImGui_ImplGlfw_SetClipboardText;
613 io.GetClipboardTextFn = ImGui_ImplGlfw_GetClipboardText;
614 io.ClipboardUserData = bd->Window;
615#ifdef __EMSCRIPTEN__
616 io.PlatformOpenInShellFn = [](ImGuiContext*, const char* url) { ImGui_ImplGlfw_EmscriptenOpenURL(url); return true; };
617#endif
618
619 // Create mouse cursors
620 // (By design, on X11 cursors are user configurable and some cursors may be missing. When a cursor doesn't exist,
621 // GLFW will emit an error which will often be printed by the app, so we temporarily disable error reporting.
622 // Missing cursors will return nullptr and our _UpdateMouseCursor() function will use the Arrow cursor instead.)
623 GLFWerrorfun prev_error_callback = glfwSetErrorCallback(nullptr);
624 bd->MouseCursors[ImGuiMouseCursor_Arrow] = glfwCreateStandardCursor(GLFW_ARROW_CURSOR);
625 bd->MouseCursors[ImGuiMouseCursor_TextInput] = glfwCreateStandardCursor(GLFW_IBEAM_CURSOR);
626 bd->MouseCursors[ImGuiMouseCursor_ResizeNS] = glfwCreateStandardCursor(GLFW_VRESIZE_CURSOR);
627 bd->MouseCursors[ImGuiMouseCursor_ResizeEW] = glfwCreateStandardCursor(GLFW_HRESIZE_CURSOR);
628 bd->MouseCursors[ImGuiMouseCursor_Hand] = glfwCreateStandardCursor(GLFW_HAND_CURSOR);
629#if GLFW_HAS_NEW_CURSORS
630 bd->MouseCursors[ImGuiMouseCursor_ResizeAll] = glfwCreateStandardCursor(GLFW_RESIZE_ALL_CURSOR);
631 bd->MouseCursors[ImGuiMouseCursor_ResizeNESW] = glfwCreateStandardCursor(GLFW_RESIZE_NESW_CURSOR);
632 bd->MouseCursors[ImGuiMouseCursor_ResizeNWSE] = glfwCreateStandardCursor(GLFW_RESIZE_NWSE_CURSOR);
633 bd->MouseCursors[ImGuiMouseCursor_NotAllowed] = glfwCreateStandardCursor(GLFW_NOT_ALLOWED_CURSOR);
634#else
635 bd->MouseCursors[ImGuiMouseCursor_ResizeAll] = glfwCreateStandardCursor(GLFW_ARROW_CURSOR);
636 bd->MouseCursors[ImGuiMouseCursor_ResizeNESW] = glfwCreateStandardCursor(GLFW_ARROW_CURSOR);
637 bd->MouseCursors[ImGuiMouseCursor_ResizeNWSE] = glfwCreateStandardCursor(GLFW_ARROW_CURSOR);
638 bd->MouseCursors[ImGuiMouseCursor_NotAllowed] = glfwCreateStandardCursor(GLFW_ARROW_CURSOR);
639#endif
640 glfwSetErrorCallback(prev_error_callback);
641#if GLFW_HAS_GETERROR && !defined(__EMSCRIPTEN__) // Eat errors (see #5908)
642 (void)glfwGetError(nullptr);
643#endif
644
645 // Chain GLFW callbacks: our callbacks will call the user's previously installed callbacks, if any.
646 if (install_callbacks)
647 ImGui_ImplGlfw_InstallCallbacks(window);
648
649 // Update monitors the first time (note: monitor callback are broken in GLFW 3.2 and earlier, see github.com/glfw/glfw/issues/784)
650 ImGui_ImplGlfw_UpdateMonitors();
651 glfwSetMonitorCallback(ImGui_ImplGlfw_MonitorCallback);
652
653 // Set platform dependent data in viewport
654 ImGuiViewport* main_viewport = ImGui::GetMainViewport();
655 main_viewport->PlatformHandle = (void*)bd->Window;
656#ifdef _WIN32
657 main_viewport->PlatformHandleRaw = glfwGetWin32Window(bd->Window);
658#elif defined(__APPLE__)
659 main_viewport->PlatformHandleRaw = (void*)glfwGetCocoaWindow(bd->Window);
660#else
661 IM_UNUSED(main_viewport);
662#endif
663 if (io.ConfigFlags & ImGuiConfigFlags_ViewportsEnable)
664 ImGui_ImplGlfw_InitPlatformInterface();
665
666 // Windows: register a WndProc hook so we can intercept some messages.
667#ifdef _WIN32
668 bd->PrevWndProc = (WNDPROC)::GetWindowLongPtrW((HWND)main_viewport->PlatformHandleRaw, GWLP_WNDPROC);
669 IM_ASSERT(bd->PrevWndProc != nullptr);
670 ::SetWindowLongPtrW((HWND)main_viewport->PlatformHandleRaw, GWLP_WNDPROC, (LONG_PTR)ImGui_ImplGlfw_WndProc);
671#endif
672
673 bd->ClientApi = client_api;
674 return true;
675}

723static void ImGui_ImplGlfw_UpdateMouseData()
724{
725 ImGui_ImplGlfw_Data* bd = ImGui_ImplGlfw_GetBackendData();
726 ImGuiIO& io = ImGui::GetIO();
727 ImGuiPlatformIO& platform_io = ImGui::GetPlatformIO();
728
729 ImGuiID mouse_viewport_id = 0;
730 const ImVec2 mouse_pos_prev = io.MousePos;
731 for (int n = 0; n < platform_io.Viewports.Size; n++)
732 {
733 ImGuiViewport* viewport = platform_io.Viewports[n];
734 GLFWwindow* window = (GLFWwindow*)viewport->PlatformHandle;
735
736#ifdef EMSCRIPTEN_USE_EMBEDDED_GLFW3
737 const bool is_window_focused = true;
738#else
739 const bool is_window_focused = glfwGetWindowAttrib(window, GLFW_FOCUSED) != 0;
740#endif
741 if (is_window_focused)
742 {
743 // (Optional) Set OS mouse position from Dear ImGui if requested (rarely used, only when ImGuiConfigFlags_NavEnableSetMousePos is enabled by user)
744 // When multi-viewports are enabled, all Dear ImGui positions are same as OS positions.
745 if (io.WantSetMousePos)
746 glfwSetCursorPos(window, (double)(mouse_pos_prev.x - viewport->Pos.x), (double)(mouse_pos_prev.y - viewport->Pos.y));
747
748 // (Optional) Fallback to provide mouse position when focused (ImGui_ImplGlfw_CursorPosCallback already provides this when hovered or captured)
749 if (bd->MouseWindow == nullptr)
750 {
751 double mouse_x, mouse_y;
752 glfwGetCursorPos(window, &mouse_x, &mouse_y);
753 if (io.ConfigFlags & ImGuiConfigFlags_ViewportsEnable)
754 {
755 // Single viewport mode: mouse position in client window coordinates (io.MousePos is (0,0) when the mouse is on the upper-left corner of the app window)
756 // Multi-viewport mode: mouse position in OS absolute coordinates (io.MousePos is (0,0) when the mouse is on the upper-left of the primary monitor)
757 int window_x, window_y;
758 glfwGetWindowPos(window, &window_x, &window_y);
759 mouse_x += window_x;
760 mouse_y += window_y;
761 }
762 bd->LastValidMousePos = ImVec2((float)mouse_x, (float)mouse_y);
763 io.AddMousePosEvent((float)mouse_x, (float)mouse_y);
764 }
765 }
766
767 // (Optional) When using multiple viewports: call io.AddMouseViewportEvent() with the viewport the OS mouse cursor is hovering.
768 // If ImGuiBackendFlags_HasMouseHoveredViewport is not set by the backend, Dear imGui will ignore this field and infer the information using its flawed heuristic.
769 // - [X] GLFW >= 3.3 backend ON WINDOWS ONLY does correctly ignore viewports with the _NoInputs flag (since we implement hit via our WndProc hook)
770 // On other platforms we rely on the library fallbacking to its own search when reporting a viewport with _NoInputs flag.
771 // - [!] GLFW <= 3.2 backend CANNOT correctly ignore viewports with the _NoInputs flag, and CANNOT reported Hovered Viewport because of mouse capture.
772 // Some backend are not able to handle that correctly. If a backend report an hovered viewport that has the _NoInputs flag (e.g. when dragging a window
773 // for docking, the viewport has the _NoInputs flag in order to allow us to find the viewport under), then Dear ImGui is forced to ignore the value reported
774 // by the backend, and use its flawed heuristic to guess the viewport behind.
775 // - [X] GLFW backend correctly reports this regardless of another viewport behind focused and dragged from (we need this to find a useful drag and drop target).
776 // FIXME: This is currently only correct on Win32. See what we do below with the WM_NCHITTEST, missing an equivalent for other systems.
777 // See https://github.com/glfw/glfw/issues/1236 if you want to help in making this a GLFW feature.
778#if GLFW_HAS_MOUSE_PASSTHROUGH
779 const bool window_no_input = (viewport->Flags & ImGuiViewportFlags_NoInputs) != 0;
780 glfwSetWindowAttrib(window, GLFW_MOUSE_PASSTHROUGH, window_no_input);
781#endif
782#if GLFW_HAS_MOUSE_PASSTHROUGH || GLFW_HAS_WINDOW_HOVERED
783 if (glfwGetWindowAttrib(window, GLFW_HOVERED))
784 mouse_viewport_id = viewport->ID;
785#else
786 // We cannot use bd->MouseWindow maintained from CursorEnter/Leave callbacks, because it is locked to the window capturing mouse.
787#endif
788 }
789
790 if (io.BackendFlags & ImGuiBackendFlags_HasMouseHoveredViewport)
791 io.AddMouseViewportEvent(mouse_viewport_id);
792}

794static void ImGui_ImplGlfw_UpdateMouseCursor()

796 ImGuiIO& io = ImGui::GetIO();
797 ImGui_ImplGlfw_Data* bd = ImGui_ImplGlfw_GetBackendData();

799 return;

811 else

814 // FIXME-PLATFORM: Unfocused windows seems to fail changing the mouse cursor with GLFW 3.2, but 3.3 works here.
815 glfwSetCursor(window, bd->MouseCursors[imgui_cursor] ? bd->MouseCursors[imgui_cursor] : bd->MouseCursors[ImGuiMouseCursor_Arrow]);
816 glfwSetInputMode(window, GLFW_CURSOR, GLFW_CURSOR_NORMAL);
817 }

874static void ImGui_ImplGlfw_UpdateMonitors()
875{
876 ImGui_ImplGlfw_Data* bd = ImGui_ImplGlfw_GetBackendData();
877 ImGuiPlatformIO& platform_io = ImGui::GetPlatformIO();
878 bd->WantUpdateMonitors = false;
879
880 int monitors_count = 0;
881 GLFWmonitor** glfw_monitors = glfwGetMonitors(&monitors_count);
882 if (monitors_count == 0) // Preserve existing monitor list if there are none. Happens on macOS sleeping (#5683)
883 return;
884
885 platform_io.Monitors.resize(0);
886 for (int n = 0; n < monitors_count; n++)
887 {
888 ImGuiPlatformMonitor monitor;
889 int x, y;
890 glfwGetMonitorPos(glfw_monitors[n], &x, &y);
891 const GLFWvidmode* vid_mode = glfwGetVideoMode(glfw_monitors[n]);
892 if (vid_mode == nullptr)
893 continue; // Failed to get Video mode (e.g. Emscripten does not support this function)
894 monitor.MainPos = monitor.WorkPos = ImVec2((float)x, (float)y);
895 monitor.MainSize = monitor.WorkSize = ImVec2((float)vid_mode->width, (float)vid_mode->height);
896#if GLFW_HAS_MONITOR_WORK_AREA
897 int w, h;
898 glfwGetMonitorWorkarea(glfw_monitors[n], &x, &y, &w, &h);
899 if (w > 0 && h > 0) // Workaround a small GLFW issue reporting zero on monitor changes: https://github.com/glfw/glfw/pull/1761
900 {
901 monitor.WorkPos = ImVec2((float)x, (float)y);
902 monitor.WorkSize = ImVec2((float)w, (float)h);
903 }
904#endif
905#if GLFW_HAS_PER_MONITOR_DPI
906 // Warning: the validity of monitor DPI information on Windows depends on the application DPI awareness settings, which generally needs to be set in the manifest or at runtime.
907 float x_scale, y_scale;
908 glfwGetMonitorContentScale(glfw_monitors[n], &x_scale, &y_scale);
909 monitor.DpiScale = x_scale;
910#endif
911 monitor.PlatformHandle = (void*)glfw_monitors[n]; // [...] GLFW doc states: "guaranteed to be valid only until the monitor configuration changes"
912 platform_io.Monitors.push_back(monitor);
913 }

916void ImGui_ImplGlfw_NewFrame()
917{
918 ImGuiIO& io = ImGui::GetIO();
919 ImGui_ImplGlfw_Data* bd = ImGui_ImplGlfw_GetBackendData();
920 IM_ASSERT(bd != nullptr && "Context or backend not initialized! Did you call ImGui_ImplGlfw_InitForXXX()?");
921
922 // Setup display size (every frame to accommodate for window resizing)
923 int w, h;
924 int display_w, display_h;
925 glfwGetWindowSize(bd->Window, &w, &h);
926 glfwGetFramebufferSize(bd->Window, &display_w, &display_h);
927 io.DisplaySize = ImVec2((float)w, (float)h);
928 if (w > 0 && h > 0)
929 io.DisplayFramebufferScale = ImVec2((float)display_w / (float)w, (float)display_h / (float)h);
930 if (bd->WantUpdateMonitors)
931 ImGui_ImplGlfw_UpdateMonitors();
932
933 // Setup time step
934 // (Accept glfwGetTime() not returning a monotonically increasing value. Seems to happens on disconnecting peripherals and probably on VMs and Emscripten, see #6491, #6189, #6114, #3644)
935 double current_time = glfwGetTime();
936 if (current_time <= bd->Time)
937 current_time = bd->Time + 0.00001f;
938 io.DeltaTime = bd->Time > 0.0 ? (float)(current_time - bd->Time) : (float)(1.0f / 60.0f);
939 bd->Time = current_time;
940
941 ImGui_ImplGlfw_UpdateMouseData();
942 ImGui_ImplGlfw_UpdateMouseCursor();
943
944 // Update game controllers (if enabled and available)
945 ImGui_ImplGlfw_UpdateGamepads();
946}
