10 Shader::Shader(const char* vertexPath, const char* fragmentPath)
11 {
12 //glfwMakeContextCurrent(PAIN::RenderStup::Windowvar);
13
14 // 1. retrieve the vertex/fragment source code from filePath
15 std::string vertexCode;
16 std::string fragmentCode;
17 std::ifstream vShaderFile;
18 std::ifstream fShaderFile;
19 // ensure ifstream objects can throw exceptions:
20 vShaderFile.exceptions(std::ifstream::failbit | std::ifstream::badbit);
21 fShaderFile.exceptions(std::ifstream::failbit | std::ifstream::badbit);
22 try
23 {
24 // open files
25 vShaderFile.open(vertexPath);
26 fShaderFile.open(fragmentPath);
27 std::stringstream vShaderStream, fShaderStream;
28 // read file's buffer contents into streams
29 vShaderStream << vShaderFile.rdbuf();
30 fShaderStream << fShaderFile.rdbuf();
31 // close file handlers
32 vShaderFile.close();
33 fShaderFile.close();
34 // convert stream into string
35 vertexCode = vShaderStream.str();
36 fragmentCode = fShaderStream.str();
37 }
38 catch (std::ifstream::failure& e)
39 {
40 std::cout << "ERROR::SHADER::FILE_NOT_SUCCESSFULLY_READ: " << e.what() << std::endl;
41 }
42 const char* vShaderCode = vertexCode.c_str();
43 const char* fShaderCode = fragmentCode.c_str();
44 // 2. compile shaders
45 unsigned int vertex, fragment;
46 // vertex shader
47 vertex = glCreateShader(GL_VERTEX_SHADER);
48 glShaderSource(vertex, 1, &vShaderCode, NULL);
49 glCompileShader(vertex);
50 checkCompileErrors(vertex, "VERTEX");
51 // fragment Shader
52 fragment = glCreateShader(GL_FRAGMENT_SHADER);
53 glShaderSource(fragment, 1, &fShaderCode, NULL);
54 glCompileShader(fragment);
55 checkCompileErrors(fragment, "FRAGMENT");
56 // shader Program
57 ID = glCreateProgram();
58 glAttachShader(ID, vertex);
59 glAttachShader(ID, fragment);
60 glLinkProgram(ID);
61 checkCompileErrors(ID, "PROGRAM");
62 // delete the shaders as they're linked into our program now and no longer necessary
63 glDeleteShader(vertex);
64 glDeleteShader(fragment);
65 //glfwMakeContextCurrent(NULL);
66 }

224 void Shader::checkCompileErrors(GLuint shader, std::string type)
225 {
226 GLint success;
227 GLchar infoLog[1024];
228 if (type != "PROGRAM")
229 {
230 glGetShaderiv(shader, GL_COMPILE_STATUS, &success);
231 if (!success)
232 {
233 glGetShaderInfoLog(shader, 1024, NULL, infoLog);
234 std::cout << "ERROR::SHADER_COMPILATION_ERROR of type: " << type << "\n" << infoLog << "\n -- --------------------------------------------------- -- " << std::endl;
235 }
236 }
237 else
238 {
239 glGetProgramiv(shader, GL_LINK_STATUS, &success);
240 if (!success)
241 {
242 glGetProgramInfoLog(shader, 1024, NULL, infoLog);
243 std::cout << "ERROR::PROGRAM_LINKING_ERROR of type: " << type << "\n" << infoLog << "\n -- --------------------------------------------------- -- " << std::endl;
244 }
245 }
