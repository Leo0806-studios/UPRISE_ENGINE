415 std::pair<const aiTexture*, int> GetEmbeddedTextureAndIndex(const char* filename) const {
416 if (nullptr==filename) {
417 return std::make_pair(nullptr, -1);
418 }
419 // lookup using texture ID (if referenced like: "*1", "*2", etc.)
420 if ('*' == *filename) {
421 int index = std::atoi(filename + 1);
422 if (0 > index || mNumTextures <= static_cast<unsigned>(index)) {
423 return std::make_pair(nullptr, -1);
424 }
425 return std::make_pair(mTextures[index], index);
426 }
427 // lookup using filename
428 const char* shortFilename = GetShortFilename(filename);
429 if (nullptr == shortFilename) {
430 return std::make_pair(nullptr, -1);
431 }
432
433 for (unsigned int i = 0; i < mNumTextures; i++) {
434 const char* shortTextureFilename = GetShortFilename(mTextures[i]->mFilename.C_Str());
435 if (strcmp(shortTextureFilename, shortFilename) == 0) {
436 return std::make_pair(mTextures[i], i);
437 }
438 }
439 return std::make_pair(nullptr, -1);
440 }
