84 // vertex positions

270 unsigned int Model::TextureFromFile(const char* path, const string& directory, bool gamma)
271 {
272 //TrPr(ctx, __func__)
273 string filename = string(path);
274 filename = directory + '\\' + filename;
275
276 unsigned int textureID;
277 glGenTextures(1, &textureID);
278
279 int width, height, nrComponents;
280 unsigned char* data = stbi_load(filename.c_str(), &width, &height, &nrComponents, 0);
281 if (data)
282 {
283 GLenum format;
284 if (nrComponents == 1)
285 format = GL_RED;
286 else if (nrComponents == 3)
287 format = GL_RGB;
288 else if (nrComponents == 4)
289 format = GL_RGBA;
290
291 glBindTexture(GL_TEXTURE_2D, textureID);
292 glTexImage2D(GL_TEXTURE_2D, 0, format, width, height, 0, format, GL_UNSIGNED_BYTE, data);
293 glGenerateMipmap(GL_TEXTURE_2D);
294
295 glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_WRAP_S, GL_REPEAT);
296 glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_WRAP_T, GL_REPEAT);
297 glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_MIN_FILTER, GL_LINEAR_MIPMAP_LINEAR);
298 glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_MAG_FILTER, GL_LINEAR);
299
300 stbi_image_free(data);
301 }
302 else
303 {
304 std::cout << "Texture failed to load at path: " << path << std::endl;
305 stbi_image_free(data);
306 }
307 //TrPrE(ctx);
