212 template <typename L>
213 void parse_code_view_records(stream_reader &stream, size_t length, L callback, size_t alignment = 1)
214 {
215 const size_t end = stream.tell() + length;
216
217 // A list of records in CodeView format
218 while (stream.tell() < end)
219 {
220 // Each records starts with 2 bytes containing the size of the record after this element
221 const auto size = stream.read<uint16_t>();
222 // Next 2 bytes contain an enumeration depicting the type and format of the following data
223 const auto code_view_tag = stream.read<uint16_t>();
224 // The next record is found by adding the current record size to the position of the previous size element
225 const auto next_record_offset = (stream.tell() - sizeof(size)) + size;
226
227 callback(code_view_tag);
228
229 stream.seek(next_record_offset);
230 stream.align(alignment);
231 }
232 }
