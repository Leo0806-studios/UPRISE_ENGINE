241AI_FORCE_INLINE
242bool aiColor4t<TReal>::IsBlack() const {
243 // The alpha component doesn't care here. black is black.
244 static const TReal epsilon = 10e-3f;
245 return std::fabs( r ) < epsilon && std::fabs( g ) < epsilon && std::fabs( b ) < epsilon;
