58TRACY_API uint32_t GetThreadHandleImpl()
59{
60#if defined _WIN32
61 static_assert( sizeof( decltype( GetCurrentThreadId() ) ) <= sizeof( uint32_t ), "Thread handle too big to fit in protocol" );
62 return uint32_t( GetCurrentThreadId() );
63#elif defined __APPLE__
64 uint64_t id;
65 pthread_threadid_np( pthread_self(), &id );
66 return uint32_t( id );
67#elif defined __ANDROID__
68 return (uint32_t)gettid();
69#elif defined __linux__
70 return (uint32_t)syscall( SYS_gettid );
71#elif defined __FreeBSD__
72 long id;
73 thr_self( &id );
74 return id;
75#elif defined __NetBSD__
76 return _lwp_self();
77#elif defined __DragonFly__
78 return lwp_gettid();
79#elif defined __OpenBSD__
80 return getthrid();
81#elif defined __EMSCRIPTEN__
82 // Not supported, but let it compile.
83 return 0;
84#else
85 // To add support for a platform, retrieve and return the kernel thread identifier here.
86 //
87 // Note that pthread_t (as for example returned by pthread_self()) is *not* a kernel
88 // thread identifier. It is a pointer to a library-allocated data structure instead.
89 // Such pointers will be reused heavily, making the pthread_t non-unique. Additionally
90 // a 64-bit pointer cannot be reliably truncated to 32 bits.
91 #error "Unsupported platform!"
92#endif

132{

154# endif

165#endif

174#else
175 pthread_setname_np( pthread_self(), buf );
176#endif
177 }
178 }
179#endif

185 buf[sz] = '\0';

189 data->next = GetThreadNameData().load( std::memory_order_relaxed );

195TRACY_API const char* GetThreadName( uint32_t id )
196{
197 static char buf[256];
198#ifdef TRACY_ENABLE
199 auto ptr = GetThreadNameData().load( std::memory_order_relaxed );
200 while( ptr )
201 {
202 if( ptr->id == id )
203 {
204 return ptr->name;
205 }
206 ptr = ptr->next;
207 }
208#endif
209
210#if defined _WIN32
211# ifdef TRACY_UWP
212 static auto _GetThreadDescription = &::GetThreadDescription;
213# else
214 static auto _GetThreadDescription = (t_GetThreadDescription)GetProcAddress( GetModuleHandleA( "kernel32.dll" ), "GetThreadDescription" );
215# endif
216 if( _GetThreadDescription )
217 {
218 auto hnd = OpenThread( THREAD_QUERY_LIMITED_INFORMATION, FALSE, (DWORD)id );
219 if( hnd != 0 )
220 {
221 PWSTR tmp;
222 if( SUCCEEDED( _GetThreadDescription( hnd, &tmp ) ) )
223 {
224 auto ret = wcstombs( buf, tmp, 256 );
225 CloseHandle( hnd );
226 LocalFree( tmp );
227 if( ret != static_cast<size_t>( -1 ) )
228 {
229 return buf;
230 }
231 }
232 }
233 }
234#elif defined __linux__
235 int cs, fd;
236 char path[32];
237 snprintf( path, sizeof( path ), "/proc/self/task/%d/comm", id );
238 sprintf( buf, "%" PRIu32, id );
239# ifndef __ANDROID__
240 pthread_setcancelstate( PTHREAD_CANCEL_DISABLE, &cs );
241# endif
242 if ( ( fd = open( path, O_RDONLY ) ) > 0) {
243 int len = read( fd, buf, 255 );
244 if( len > 0 )
245 {
246 buf[len] = 0;
247 if( len > 1 && buf[len-1] == '\n' )
248 {
249 buf[len-1] = 0;
250 }
251 }
252 close( fd );
253 }
254# ifndef __ANDROID__
255 pthread_setcancelstate( cs, 0 );
256# endif
257 return buf;
258#endif
259
260 sprintf( buf, "%" PRIu32, id );
261 return buf;
262}

264TRACY_API const char* GetEnvVar( const char* name )
265{
266#if defined _WIN32
267 // unfortunately getenv() on Windows is just fundamentally broken. It caches the entire
268 // environment block once on startup, then never refreshes it again. If any environment
269 // strings are added or modified after startup of the CRT, those changes will not be
270 // seen by getenv(). This removes the possibility of an app using this SDK from
271 // programmatically setting any of the behaviour controlling envvars here.
272 //
273 // To work around this, we'll instead go directly to the Win32 environment strings APIs
274 // to get the current value.
275 static char buffer[1024];
276 DWORD const kBufferSize = DWORD(sizeof(buffer) / sizeof(buffer[0]));
277 DWORD count = GetEnvironmentVariableA(name, buffer, kBufferSize);
278
279 if( count == 0 )
280 return nullptr;
281
282 if( count >= kBufferSize )
283 {
284 char* buf = reinterpret_cast<char*>(_alloca(count + 1));
285 count = GetEnvironmentVariableA(name, buf, count + 1);
286 memcpy(buffer, buf, kBufferSize);
287 buffer[kBufferSize - 1] = 0;
288 }
289
290 return buffer;
291#else
292 return getenv(name);
293#endif
294}
