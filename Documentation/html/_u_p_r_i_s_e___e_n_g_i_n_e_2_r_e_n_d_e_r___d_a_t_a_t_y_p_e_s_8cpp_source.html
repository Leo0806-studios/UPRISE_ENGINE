72 void Mesh::setupMesh()
73 {
74 glGenVertexArrays(1, &VAO);
75 glGenBuffers(1, &VBO);
76 glGenBuffers(1, &EBO);
77 glBindVertexArray(VAO);
78 glBindBuffer(GL_ARRAY_BUFFER, VBO);

82 glBufferData(GL_ELEMENT_ARRAY_BUFFER, indices.size() *

84 // vertex positions
85 glEnableVertexAttribArray(0);

270 unsigned int Model::TextureFromFile(const char* path, const string& directory, bool gamma)
271 {
272 //TrPr(ctx, __func__)

274 filename = directory + '\\' + filename;
275
276 unsigned int textureID;
277 glGenTextures(1, &textureID);

281 if (data)

283 GLenum format;

298 glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_MAG_FILTER, GL_LINEAR);
