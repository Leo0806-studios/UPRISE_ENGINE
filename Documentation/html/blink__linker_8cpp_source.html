13static void write_jump(uint8_t *address, const uint8_t *jump_target)
14{
15#ifdef _M_IX86
16 DWORD protect = PAGE_READWRITE;
17 VirtualProtect(address, 5, protect, &protect);
18
19 // JMP
20 address[0] = 0xE9;
21 *reinterpret_cast<int32_t *>(address + 1) = jump_target - (address + 5);

23 VirtualProtect(address, 5, protect, &protect);
24#endif
25#ifdef _M_AMD64
26 DWORD protect = PAGE_READWRITE;
27 VirtualProtect(address, 12, protect, &protect);
28
29 // MOV RAX, [target_address]
30 // JMP RAX
31 address[0] = 0x48;

34 address[10] = 0xFF;

38#endif
39}

41static uint8_t *find_free_memory_region(uint8_t *address, size_t size)
42{
43#ifdef _M_AMD64
44 SYSTEM_INFO sysinfo;
45 MEMORY_BASIC_INFORMATION meminfo;
46 GetSystemInfo(&sysinfo);
47
48 address -= reinterpret_cast<uintptr_t>(address) % sysinfo.dwAllocationGranularity;
49 address += sysinfo.dwAllocationGranularity;
50 auto maxaddress = static_cast<uint8_t *>(sysinfo.lpMaximumApplicationAddress);
51 maxaddress -= size;
52
53 while (address < maxaddress)
54 {
55 if (VirtualQuery(address, &meminfo, sizeof(meminfo)) == 0)
56 break;
57
58 if (meminfo.State == MEM_FREE)
59 return address;
60
61 address = static_cast<uint8_t *>(meminfo.BaseAddress) + meminfo.RegionSize;
62
63 // Round up to the next allocation granularity
64 address += sysinfo.dwAllocationGranularity - 1;
65 address -= reinterpret_cast<uintptr_t>(address) % sysinfo.dwAllocationGranularity;
66 }
67#endif
68 return nullptr;
69}

126 // Object file can be a normal COFF or an extended COFF
127 COFF_HEADER header;
128 const scoped_handle file = open_coff_file(path, header);
129 if (file == INVALID_HANDLE_VALUE)
130 return false;
131
132 return !header.is_extended() ?

134 link<IMAGE_SYMBOL_EX>(file, header.bigobj);
