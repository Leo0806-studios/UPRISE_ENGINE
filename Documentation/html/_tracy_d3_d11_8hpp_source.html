60 D3D11Ctx( ID3D11Device* device, ID3D11DeviceContext* devicectx )
61 {
62 // TODO: consider calling ID3D11Device::GetImmediateContext() instead of passing it as an argument
63 m_device = device;
64 device->AddRef();
65 m_immediateDevCtx = devicectx;
66 devicectx->AddRef();
67
68 {
69 D3D11_QUERY_DESC desc = { };
70 desc.Query = D3D11_QUERY_TIMESTAMP_DISJOINT;
71 if (FAILED(m_device->CreateQuery(&desc, &m_disjointQuery)))
72 {
73 TracyD3D11Panic("unable to create disjoint timestamp query.", return);
74 }
75 }
76
77 for (ID3D11Query*& query : m_queries)
78 {
79 D3D11_QUERY_DESC desc = { };
80 desc.Query = D3D11_QUERY_TIMESTAMP;
81 if (FAILED(m_device->CreateQuery(&desc, &query)))
82 {
83 TracyD3D11Panic("unable to create timestamp query.", return);
84 }
85 }
86
87 // Calibrate CPU and GPU timestamps
88 int64_t tcpu = 0;
89 int64_t tgpu = 0;
90 for (int attempts = 0; attempts < 50; attempts++)
91 {
92 m_immediateDevCtx->Begin(m_disjointQuery);
93 m_immediateDevCtx->End(m_queries[0]);
94 m_immediateDevCtx->End(m_disjointQuery);
95
96 int64_t tcpu0 = Profiler::GetTime();
97 WaitForQuery(m_disjointQuery);
98 int64_t tcpu1 = Profiler::GetTime();
99
100 D3D11_QUERY_DATA_TIMESTAMP_DISJOINT disjoint = { };
101 if (m_immediateDevCtx->GetData(m_disjointQuery, &disjoint, sizeof(disjoint), 0) != S_OK)
102 {
103 TracyMessageLC("TracyD3D11: unable to query GPU timestamp; retrying...", tracy::Color::Tomato);
104 continue;
105 }
106
107 if (disjoint.Disjoint)
108 continue;
109
110 UINT64 timestamp = 0;
111 if (m_immediateDevCtx->GetData(m_queries[0], &timestamp, sizeof(timestamp), 0) != S_OK)
112 continue; // this should never happen, since the enclosing disjoint query succeeded
113
114 tcpu = tcpu0 + (tcpu1 - tcpu0) * 1 / 2;
115 tgpu = timestamp * (1000000000 / disjoint.Frequency);
116 break;
117 }
118
119 // ready to roll
120 m_contextId = GetGpuCtxCounter().fetch_add(1);
121 m_immediateDevCtx->Begin(m_disjointQuery);
122 m_previousCheckpoint = m_nextCheckpoint = 0;
123
124 auto* item = Profiler::QueueSerial();
125 MemWrite( &item->hdr.type, QueueType::GpuNewContext );
126 MemWrite( &item->gpuNewContext.cpuTime, tcpu );
127 MemWrite( &item->gpuNewContext.gpuTime, tgpu );
128 MemWrite( &item->gpuNewContext.thread, uint32_t(0) ); // #TODO: why not GetThreadHandle()?
129 MemWrite( &item->gpuNewContext.period, 1.0f );
130 MemWrite( &item->gpuNewContext.context, m_contextId);
131 MemWrite( &item->gpuNewContext.flags, uint8_t(0) );
132 MemWrite( &item->gpuNewContext.type, GpuContextType::Direct3D11 );
133
134#ifdef TRACY_ON_DEMAND
135 GetProfiler().DeferItem( *item );
136#endif
137
138 Profiler::QueueSerialFinish();
139 }

175 void Collect(CollectMode mode = POLL)
176 {
177 ZoneScopedC( Color::Red4 );
178
179#ifdef TRACY_ON_DEMAND
180 if( !GetProfiler().IsConnected() )
181 {
182 m_previousCheckpoint = m_nextCheckpoint = m_queryCounter;
183 return;
184 }
185#endif
186
187 if (m_previousCheckpoint == m_nextCheckpoint)
188 {
189 uintptr_t nextCheckpoint = m_queryCounter;
190 if (nextCheckpoint == m_nextCheckpoint)
191 {
192 return;
193 }
194 m_nextCheckpoint = nextCheckpoint;
195 m_immediateDevCtx->End(m_disjointQuery);
196 }
197
198 if (mode == CollectMode::BLOCK)
199 {
200 WaitForQuery(m_disjointQuery);
201 }
202
203 D3D11_QUERY_DATA_TIMESTAMP_DISJOINT disjoint = { };
204 if (m_immediateDevCtx->GetData(m_disjointQuery, &disjoint, sizeof(disjoint), D3D11_ASYNC_GETDATA_DONOTFLUSH) != S_OK)
205 {
206 return;
207 }
208
209 if (disjoint.Disjoint == TRUE)
210 {
211 m_previousCheckpoint = m_nextCheckpoint;
212 TracyD3D11Panic("disjoint timestamps detected; dropping.");
213 return;
214 }
215
216 auto begin = m_previousCheckpoint;
217 auto end = m_nextCheckpoint;
218 for (auto i = begin; i != end; ++i)
219 {
220 uint32_t k = RingIndex(i);
221 UINT64 timestamp = 0;
222 if (m_immediateDevCtx->GetData(m_queries[k], &timestamp, sizeof(timestamp), 0) != S_OK)
223 {
224 TracyD3D11Panic("timestamp expected to be ready, but it was not!");
225 break;
226 }
227 timestamp *= (1000000000ull / disjoint.Frequency);
228 auto* item = Profiler::QueueSerial();
229 MemWrite(&item->hdr.type, QueueType::GpuTime);
230 MemWrite(&item->gpuTime.gpuTime, static_cast<int64_t>(timestamp));
231 MemWrite(&item->gpuTime.queryId, static_cast<uint16_t>(k));
232 MemWrite(&item->gpuTime.context, m_contextId);
233 Profiler::QueueSerialFinish();
234 }
235
236 // disjoint timestamp queries should only be invoked once per frame or less
237 // https://learn.microsoft.com/en-us/windows/win32/api/d3d11/ne-d3d11-d3d11_query
238 m_immediateDevCtx->Begin(m_disjointQuery);
239 m_previousCheckpoint = m_nextCheckpoint;
240 }
