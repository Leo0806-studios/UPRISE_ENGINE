333void blink::application::read_import_address_table(const BYTE *image_base)
334{
335 const auto headers = reinterpret_cast<const IMAGE_NT_HEADERS *>(
336 _image_base + reinterpret_cast<const IMAGE_DOS_HEADER *>(_image_base)->e_lfanew);
337
338 // Search import directory for additional symbols
339 const IMAGE_DATA_DIRECTORY &import_directory = headers->OptionalHeader.DataDirectory[IMAGE_DIRECTORY_ENTRY_IMPORT];
340 const auto import_directory_entries = reinterpret_cast<const IMAGE_IMPORT_DESCRIPTOR *>(_image_base + import_directory.VirtualAddress);
341
342 for (unsigned int i = 0; import_directory_entries[i].FirstThunk != 0; i++)
343 {
344 const auto name = reinterpret_cast<const char *>(_image_base + import_directory_entries[i].Name);
345 const auto import_name_table = reinterpret_cast<const IMAGE_THUNK_DATA *>(_image_base + import_directory_entries[i].Characteristics);
346 const auto import_address_table = reinterpret_cast<const IMAGE_THUNK_DATA *>(_image_base + import_directory_entries[i].FirstThunk);
347
348 // The module should have already been loaded by Windows when the application was launched, so just get its handle here
349 const auto target_base = reinterpret_cast<const BYTE *>(GetModuleHandleA(name));
350 if (target_base == nullptr)
351 continue; // Bail out if that is not the case to be safe
352
353 for (unsigned int k = 0; import_name_table[k].u1.AddressOfData != 0; k++)
354 {
355 const char *import_name = nullptr;
356
357 // We need to figure out the name of symbols imported by ordinal by going through the export table of the target module
358 if (IMAGE_SNAP_BY_ORDINAL(import_name_table[k].u1.Ordinal))
359 {
360 const auto target_headers = reinterpret_cast<const IMAGE_NT_HEADERS *>(target_base + reinterpret_cast<const IMAGE_DOS_HEADER *>(target_base)->e_lfanew);
361 const auto export_directory = reinterpret_cast<const IMAGE_EXPORT_DIRECTORY *>(target_base + target_headers->OptionalHeader.DataDirectory[IMAGE_DIRECTORY_ENTRY_EXPORT].VirtualAddress);
362 const auto export_name_strings = reinterpret_cast<const DWORD *>(target_base + export_directory->AddressOfNames);
363 const auto export_name_ordinals = reinterpret_cast<const WORD *>(target_base + export_directory->AddressOfNameOrdinals);
364
365 const auto ordinal = std::find(export_name_ordinals, export_name_ordinals + export_directory->NumberOfNames, IMAGE_ORDINAL(import_name_table[k].u1.Ordinal));
366 if (ordinal != export_name_ordinals + export_directory->NumberOfNames)
367 import_name = reinterpret_cast<const char *>(target_base + export_name_strings[std::distance(export_name_ordinals, ordinal)]);
368 else
369 continue; // Ignore ordinal imports for which the name could not be resolved
370 }
371 else
372 {
373 import_name = reinterpret_cast<const IMAGE_IMPORT_BY_NAME *>(_image_base + import_name_table[k].u1.AddressOfData)->Name;
374 }
375
376 _symbols.insert({ import_name, reinterpret_cast<void *>(import_address_table[k].u1.AddressOfData) });
377 }
378
379 read_debug_info(target_base);
380 }
